///// ---------- CONFIG ----------
const FACE_EAR_THRESH = 0.18;        // ear threshold for blink detection
const BLINK_DEBOUNCE_MS = 250;
const VAD_ENERGY_THRESH = 0.01;      // energy threshold for voice activity (tweak)
const VAD_MIN_SILENCE_MS = 300;      // silence threshold to count pauses

const AGG_WINDOW_MS = 5000;          // internal rolling window for some features

///// ---------- STATE ----------
let faceScore = 0;
let blinkCount = 0;
let lastBlinkTime = 0;
let blinkTimestamps = [];            // to compute blink rate
let earLeftBuf = [];
let earRightBuf = [];
let mouthBuf = [];

let typingEvents = [];               // timestamps (ms)
let backspaceCount = 0;

let audioContext = null;
let analyser = null;
let microphoneStream = null;
let audioDataArray = null;
let isVoiceRunning = false;
let pitchHz = 0;
let voiceEnergyBuf = [];             // RMS buffer
let speechActiveSamples = 0;
let totalVoiceSamples = 0;

const videoElement = document.getElementById('video');

///// ---------- TYPING ----------
const typingBox = document.getElementById('typingBox');
typingBox.addEventListener('keydown', (e) => {
  typingEvents.push(Date.now());
  if (e.key === 'Backspace') backspaceCount += 1;
});
function computeTypingScore() {
  if (typingEvents.length < 2) return 10;
  let deltas = [];
  for (let i = 1; i < typingEvents.length; i++) {
    deltas.push(typingEvents[i] - typingEvents[i-1]);
  }
  // average inter-key ms
  const avg = deltas.reduce((a,b)=>a+b,0) / deltas.length;
  // backspace rate per 100 keys
  const bsRate = (backspaceCount / Math.max(typingEvents.length,1)) * 100;
  // Score scale: faster typing -> higher score; more backspaces -> lower
  let speedScore = Math.max(0, (300/avg) * 100); // rough mapping
  speedScore = Math.min(100, speedScore);
  let penalty = Math.min(50, bsRate); // penalize backspaces
  const typingScore = Math.max(0, speedScore - penalty);
  return Math.round(typingScore);
}

///// ---------- FACE (MediaPipe FaceMesh) ----------
const faceScoreEl = document.getElementById('faceScore');
const blinkRateEl = document.getElementById('blinkRate');

// helper: euclidean
function euclid(a,b){ return Math.hypot(a[0]-b[0], a[1]-b[1]); }

// indices for EAR (MediaPipe)
const LEFT_EYE = [33,160,158,133,153,144];
const RIGHT_EYE = [362,385,387,263,373,380];

function computeEAR(landmarks, idxs){
  try {
    const p = idxs.map(i => [landmarks[i].x, landmarks[i].y]);
    const A = euclid(p[1], p[5]);
    const B = euclid(p[2], p[4]);
    const C = euclid(p[0], p[3]);
    if (C === 0) return 0;
    return (A + B) / (2.0 * C);
  } catch (e) {
    return 0;
  }
}

function computeMouthOpen(landmarks){
  // use landmarks 13 (upper lip) and 14 (lower lip) approximations
  if (!landmarks[13] || !landmarks[14]) return 0;
  return euclid([landmarks[13].x, landmarks[13].y],[landmarks[14].x, landmarks[14].y]);
}

// MediaPipe setup
const faceMesh = new FaceMesh({
  locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${file}`
});
faceMesh.setOptions({
  maxNumFaces: 1,
  refineLandmarks: true,
  minDetectionConfidence: 0.5,
  minTrackingConfidence: 0.5
});
faceMesh.onResults(onFaceResults);

let mpCamera = null;
async function startCamera() {
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ video: { width:320, height:240 } });
    videoElement.srcObject = stream;
    // feed MediaPipe using Camera utils
    mpCamera = new Camera(videoElement, {
      onFrame: async () => {
        await faceMesh.send({image: videoElement});
      },
      width: 320,
      height: 240
    });
    mpCamera.start();
  } catch (err) {
    console.warn('Camera permission or device error:', err);
  }
}

function onFaceResults(results) {
  if (!results.multiFaceLandmarks || results.multiFaceLandmarks.length === 0) {
    // no face
    return;
  }
  const lm = results.multiFaceLandmarks[0]; // landmarks with x,y,z properties normalized
  // compute EARs
  const earL = computeEAR(lm, LEFT_EYE);
  const earR = computeEAR(lm, RIGHT_EYE);
  const earAvg = (earL + earR) / 2.0;
  const mouthOpen = computeMouthOpen(lm);

  // append to rolling buffers
  const now = Date.now();
  earLeftBuf.push({t:now, v:earL}); if (earLeftBuf.length > 200) earLeftBuf.shift();
  earRightBuf.push({t:now, v:earR}); if (earRightBuf.length > 200) earRightBuf.shift();
  mouthBuf.push({t:now, v:mouthOpen}); if (mouthBuf.length > 200) mouthBuf.shift();

  // blink detection (simple threshold + debounce)
  if (earAvg < FACE_EAR_THRESH) {
    if (now - lastBlinkTime > BLINK_DEBOUNCE_MS) {
      blinkCount += 1;
      blinkTimestamps.push(now);
      lastBlinkTime = now;
      // prune old
      const windowMs = 60_000 * 5;
      blinkTimestamps = blinkTimestamps.filter(ts => ts > now - windowMs);
    }
  }

  // compute faceScore (simple heuristic): low EAR variance + high mouth variance => possible stress/fatigue
  const earValues = earLeftBuf.map(x=>x.v).concat(earRightBuf.map(x=>x.v));
  const earMean = earValues.reduce((a,b)=>a+b,0) / Math.max(1,earValues.length);
  const earStd = Math.sqrt(earValues.reduce((s,x)=>s+(x-earMean)*(x-earMean),0) / Math.max(1,earValues.length));
  const mouthVals = mouthBuf.map(x=>x.v);
  const mouthStd = Math.sqrt(mouthVals.reduce((s,x)=>s+(x-(mouthVals.reduce((a,b)=>a+b,0)/Math.max(1,mouthVals.length)))**2,0)/Math.max(1,mouthVals.length) || 0);

  // Map metrics into 0-100
  // Lower EAR mean (more closed eyes) and higher blinkRate indicate fatigue/strain
  const blinkRatePerMin = (blinkTimestamps.length / Math.max(1,( (Date.now() - Math.max(...blinkTimestamps,0)) / 60000))) || (blinkTimestamps.length);
  // However to avoid divide by zero, compute blink/min over last minute
  const oneMinAgo = Date.now() - 60000;
  const blinksLastMin = blinkTimestamps.filter(ts => ts > oneMinAgo).length;
  const blinkRate = blinksLastMin; // per minute approx

  // Heuristic face score:
  // low EARMean and larger mouth std and higher blink rate increases score
  let faceScoreLocal = 0;
  faceScoreLocal += Math.max(0, (0.25 - earMean) * 300);   // EAR mean mapping
  faceScoreLocal += Math.min(40, mouthStd * 200);          // mouth variance mapping
  faceScoreLocal += Math.min(40, blinkRate * 3);           // blink mapping
  faceScoreLocal = Math.min(100, Math.max(0, Math.round(faceScoreLocal)));
  faceScore = faceScoreLocal;

  // Update UI
  faceScoreEl.innerText = faceScore;
  blinkRateEl.innerText = blinksLastMin;
}

///// ---------- VOICE (WebAudio) ----------
const voiceToggle = document.getElementById('voiceToggle');
const micConsent = document.getElementById('micConsent');
const voiceScoreEl = document.getElementById('voiceScore');
const pitchHzEl = document.getElementById('pitchHz');
const speechPercentEl = document.getElementById('speechPercent');

voiceToggle.onclick = async () => {
  if (!isVoiceRunning) {
    if (!micConsent.checked) {
      alert("Please check 'Allow mic' to enable voice monitoring.");
      return;
    }
    await startVoiceProcessing();
    voiceToggle.innerText = 'Stop voice';
  } else {
    stopVoiceProcessing();
    voiceToggle.innerText = 'Start voice';
  }
};

async function startVoiceProcessing(){
  try {
    audioContext = new (window.AudioContext || window.webkitAudioContext)();
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    microphoneStream = stream;
    const source = audioContext.createMediaStreamSource(stream);
    analyser = audioContext.createAnalyser();
    analyser.fftSize = 2048;
    audioDataArray = new Float32Array(analyser.fftSize);
    source.connect(analyser);
    isVoiceRunning = true;
    voiceProcessingLoop();
  } catch (e) {
    console.warn("Microphone access error:", e);
  }
}

function stopVoiceProcessing(){
  isVoiceRunning = false;
  try {
    if (microphoneStream) {
      microphoneStream.getTracks().forEach(t=>t.stop());
      microphoneStream = null;
    }
    if (audioContext) {
      audioContext.close();
      audioContext = null;
    }
  } catch(e){}
}

// compute RMS energy
function computeRMS(buf){
  let sum=0;
  for(let i=0;i<buf.length;i++){ sum += buf[i]*buf[i]; }
  return Math.sqrt(sum / buf.length);
}

// simple autocorrelation pitch detection (returns freq in Hz or 0)
function detectPitch(buf, sr) {
  // buffer should be Float32Array
  let SIZE = buf.length;
  let rms = computeRMS(buf);
  if (rms < 0.002) return 0; // silence
  let r = new Array(SIZE).fill(0);
  for (let lag=0; lag < SIZE; lag++){
    let sum = 0;
    for (let i=0; i < SIZE - lag; i++){
      sum += buf[i] * buf[i+lag];
    }
    r[lag] = sum;
  }
  // find peak in r after lag 20
  let peakIndex = -1, peakVal = 0;
  for (let i=20; i<r.length; i++){
    if (r[i] > peakVal) { peakVal = r[i]; peakIndex = i; }
  }
  if (peakIndex <= 0) return 0;
  const fundamentalFreq = sr / peakIndex;
  if (fundamentalFreq > 50 && fundamentalFreq < 1000) return Math.round(fundamentalFreq);
  return 0;
}

async function voiceProcessingLoop(){
  const sr = audioContext.sampleRate;
  const buf = new Float32Array(analyser.fftSize);
  const lastActiveTimes = [];
  while(isVoiceRunning){
    analyser.getFloatTimeDomainData(buf);
    const rms = computeRMS(buf);
    voiceEnergyBuf.push(rms);
    if (voiceEnergyBuf.length > 200) voiceEnergyBuf.shift();
    // VAD: consider voiced if rms > threshold
    const voiced = (rms > VAD_ENERGY_THRESH) ? 1 : 0;
    if (voiced) speechActiveSamples++;
    totalVoiceSamples++;

    // compute pitch every ~200ms
    const p = detectPitch(buf, sr);
    if (p > 0) pitchHz = p;

    // update UI metrics
    const energyAvg = voiceEnergyBuf.reduce((a,b)=>a+b,0)/Math.max(1,voiceEnergyBuf.length);
    // voice score mapping: higher energy variation + pitch instability -> higher score
    let energyScore = Math.min(60, Math.round(Math.min(1, energyAvg * 1000) * 60));
    let pitchScore = (pitchHz === 0) ? 0 : Math.min(40, Math.round( Math.abs(220 - pitchHz) / 2 ));
    let voiceScoreLocal = Math.min(100, energyScore + pitchScore);
    voiceScoreEl.innerText = voiceScoreLocal;
    pitchHzEl.innerText = pitchHz;
    const speechPerc = Math.round((speechActiveSamples / Math.max(1,totalVoiceSamples)) * 100);
    speechPercentEl.innerText = speechPerc;

    // set global var
    window._voiceScoreLocal = voiceScoreLocal;

    await new Promise(r => setTimeout(r, 180));
  }
}

///// ---------- ANALYZE / FUSION & UI ----------
const analyzeBtn = document.getElementById('analyzeBtn');
const meterFill = document.getElementById('meterFill');
const stressValue = document.getElementById('stressValue');
const statusEl = document.getElementById('status');
const clearBtn = document.getElementById('clearBtn');

analyzeBtn.onclick = async () => {
  statusEl.innerText = "Preparing features...";
  const typingScore = computeTypingScore();
  const face = faceScore || 0;
  const voice = window._voiceScoreLocal || 0;

  // Build numeric-only payload (no text/audio/video)
  const payload = {
    typing_score: typingScore,
    facial_score: Math.round(face),
    voice_score: Math.round(voice)
  };

  statusEl.innerText = "Sending numeric features to backend (no media or text)...";
  try {
    const res = await fetch('http://127.0.0.1:5000/api/analyze', {
      method: 'POST',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify(payload)
    });
    const j = await res.json();
    const s = Math.round((j.stress_score || 0) * 100);
    stressValue.innerText = s;
    meterFill.style.width = `${s}%`;
    statusEl.innerText = "Done — score computed. No data stored by backend.";
  } catch (err) {
    console.warn(err);
    statusEl.innerText = "Could not reach backend — ensure Flask server is running on port 5000.";
  }
};

clearBtn.onclick = () => {
  // wipe local buffers
  typingEvents = []; backspaceCount = 0;
  earLeftBuf = []; earRightBuf = []; mouthBuf = []; blinkTimestamps = []; blinkCount = 0;
  voiceEnergyBuf = []; speechActiveSamples = 0; totalVoiceSamples = 0; pitchHz = 0;
  document.getElementById('typingScore').innerText = 0;
  faceScoreEl.innerText = 0; voiceScoreEl.innerText = 0; pitchHzEl.innerText = 0;
  window._voiceScoreLocal = 0;
  statusEl.innerText = 'Local buffers cleared.';
};
